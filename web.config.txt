<?xml version="1.0" encoding="UTF-8"?>
<configuration>
   <location path=".">
   <system.webServer>
       <directoryBrowse enabled="false" />
       <rewrite>
           <rules>
               <rule name="Joomla! Common Exploits Prevention" stopProcessing="true">
                   <match url="^(.*)$" ignoreCase="false" />
                   <conditions logicalGrouping="MatchAny">
                       <add input="{QUERY_STRING}" pattern="base64_encode[^(]*\([^)]*\)" ignoreCase="false" />
                       <add input="{QUERY_STRING}" pattern="(&gt;|%3C)([^s]*s)+cript.*(&lt;|%3E)" />
                       <add input="{QUERY_STRING}" pattern="GLOBALS(=|\[|\%[0-9A-Z]{0,2})" ignoreCase="false" />
                       <add input="{QUERY_STRING}" pattern="_REQUEST(=|\[|\%[0-9A-Z]{0,2})" ignoreCase="false" />
                   </conditions>
                   <action type="CustomResponse" url="index.php" statusCode="403" statusReason="Forbidden" statusDescription="Forbidden" />
               </rule>
               <!-- ################################################################### -->
               <!-- eMundus protected downloads.                                        -->
               <!-- Mirrors the RewriteRules in .htaccess.tchooz.base.rules.txt. These   -->
               <!-- MUST stay ahead of the Joomla SEF rules below: those skip any        -->
               <!-- request that resolves to an existing file, so without these rules    -->
               <!-- IIS would serve the uploads statically and bypass the PHP access     -->
               <!-- checks in EmundusController::getfile().                              -->
               <!-- ################################################################### -->
               <rule name="eMundus applicant files" stopProcessing="true">
                   <match url="^(images/emundus/files)/([0-9]+)/(.*)\.(.*)$" ignoreCase="true" />
                   <action type="Redirect" url="/index.php?option=com_emundus&amp;task=getfile&amp;u={R:1}/{R:2}/{R:3}.{R:4}" appendQueryString="false" redirectType="Found" />
               </rule>
               <rule name="eMundus exports" stopProcessing="true">
                   <match url="^(images/emundus/exports)/([0-9]+)/(.*)\.(.*)$" ignoreCase="true" />
                   <action type="Redirect" url="/index.php?option=com_emundus&amp;task=getfile&amp;u={R:1}/{R:2}/{R:3}.{R:4}" appendQueryString="false" redirectType="Found" />
               </rule>
               <rule name="eMundus file archives" stopProcessing="true">
                   <match url="^(images/emundus/files/archives)/(.*)\.(.*)$" ignoreCase="true" />
                   <action type="Redirect" url="/index.php?option=com_emundus&amp;task=getfile&amp;u={R:1}/{R:2}.{R:3}" appendQueryString="false" redirectType="Found" />
               </rule>
               <rule name="eMundus resources" stopProcessing="true">
                   <match url="^(images/emundus/resources)/(.*)\.(.*)$" ignoreCase="true" />
                   <action type="Redirect" url="/index.php?option=com_emundus&amp;task=getfile&amp;u={R:1}/{R:2}.{R:3}" appendQueryString="false" redirectType="Found" />
               </rule>
               <rule name="eMundus tmp" stopProcessing="true">
                   <match url="^(tmp)/(.*)\.(.*)$" ignoreCase="true" />
                   <action type="Redirect" url="/index.php?option=com_emundus&amp;task=getfile&amp;u={R:1}/{R:2}.{R:3}" appendQueryString="false" redirectType="Found" />
               </rule>
               <rule name="eMundus logs" stopProcessing="true">
                   <match url="^(logs)/(.*)\.(.*)$" ignoreCase="true" />
                   <action type="Redirect" url="/index.php?option=com_emundus&amp;task=getfile&amp;u={R:1}/{R:2}.{R:3}" appendQueryString="false" redirectType="Found" />
               </rule>
               <rule name="Joomla! API Application SEF URLs">
                   <match url="^api/(.*)" ignoreCase="false" />
                   <conditions logicalGrouping="MatchAll">
                     <add input="{URL}" pattern="^/api/index.php" ignoreCase="true" negate="true" />
                     <add input="{REQUEST_FILENAME}" matchType="IsFile" ignoreCase="false" negate="true" />
                     <add input="{REQUEST_FILENAME}" matchType="IsDirectory" ignoreCase="false" negate="true" />
                   </conditions>
                   <action type="Rewrite" url="api/index.php" />
               </rule>
               <rule name="Joomla! Public Frontend SEF URLs">
                   <match url="(.*)" ignoreCase="false" />
                   <conditions logicalGrouping="MatchAll">
                     <add input="{URL}" pattern="^/index.php" ignoreCase="true" negate="true" />
                     <add input="{REQUEST_FILENAME}" matchType="IsFile" ignoreCase="false" negate="true" />
                     <add input="{REQUEST_FILENAME}" matchType="IsDirectory" ignoreCase="false" negate="true" />
                   </conditions>
                   <action type="Rewrite" url="index.php" />
               </rule>
           </rules>
       </rewrite>
       <httpProtocol>
           <customHeaders>
               <add name="X-Content-Type-Options" value="nosniff" />
               <!-- Protect against certain cross-origin requests. More information can be found here: -->
               <!-- https://developer.mozilla.org/en-US/docs/Web/HTTP/Cross-Origin_Resource_Policy_(CORP) -->
               <!-- https://web.dev/why-coop-coep/ -->
               <!-- <add name="Cross-Origin-Resource-Policy" value="same-origin" /> -->
               <!-- <add name="Cross-Origin-Embedder-Policy" value="require-corp" /> -->
           </customHeaders>
       </httpProtocol>
   </system.webServer>
   </location>
</configuration>
